          root.classList.add('is-reels-loading');
        }
        loadLatestReel.retryTimer = window.setTimeout(loadLatestReel, 2500);
      } finally {
        loadLatestReel.pending = null;
      }
    })();
    return loadLatestReel.pending;
  }

    function reelSource(reel, purpose) {
      if(!reel)return '';
      const immediate = reelImmediatePlaybackUrls.get(String(reel.id || '')) || String(reel.cachedVideo || '');

    function clearVideoSource(cell) {
      const video = cellVideo(cell);
      if (!video || !video.getAttribute('src')) return;
      try { video.pause(); } catch (_error) {}
      video.removeAttribute('src');
      video.removeAttribute('poster');
      try { video.load(); } catch (_error) {}
      video.dataset.loadedSource = '';
      video.dataset.nativeGeneration = String((Number(video.dataset.nativeGeneration) || 0) + 1);
      cell.dataset.frameReady = 'false';
      video.dataset.neighborWarm = '';
      const poster = ensurePoster(cell);
      /* Posters are tiny immutable JPEGs. Keep an already-loaded poster in the
         DOM when releasing a far video so reverse scrolling never waits for the
         thumbnail network/decode path again. */
      if (!poster.getAttribute('src') && poster.dataset.posterSrc) poster.src = poster.dataset.posterSrc;
      poster.hidden = !poster.getAttribute('src');
    }

    function loadVideoSource(cell, preloadMode, sourceOverride) {
      if (!cell || !cell.__reel) return null;
      const video = cellVideo(cell);
      if (!video) return null;
      const source = String(sourceOverride || reelSource(cell.__reel));
      if (!source) return video;
      video.preload = preloadMode || 'auto';
      if (video.dataset.loadedSource !== source || video.getAttribute('src') !== source) {
        try { video.pause(); } catch (_error) {}
        const generation = String((Number(video.dataset.nativeGeneration) || 0) + 1);
        video.dataset.nativeGeneration = generation;
        video.dataset.loadedSource = source;
        cell.dataset.frameReady = 'false';
        if (!video.dataset.neighborWarm) video.dataset.neighborWarm = '';
        video.classList.add('is-awaiting-frame');
        const poster = ensurePoster(cell);
        primeCellPoster(cell);
        video.removeAttribute('poster');
        video.src = source;
        try { video.load(); } catch (_error) {}
      }
      return video;
    }

    function preloadNeighbor(cell, mode) {
      if (!cell || !cell.__reel || cell.classList.contains('is-active')) return;
      const source = reelSource(cell.__reel, 'warm');
      if (!source) return;
      const decode = mode === 'decode';
      const video = cellVideo(cell);
      if (!video) return;
      video.dataset.neighborWarm = decode ? 'decode' : 'metadata';
      primeCellPoster(cell);
      /* IMPORTANT: preload the exact URL that will later play. Media-fragment
         URLs such as #t=0,3 create a distinct currentSrc in Chromium, which can
         make activation reload the media and discard the prepared buffer. */
      const loaded = loadVideoSource(cell, decode ? 'auto' : 'metadata', source);
      if (!loaded) return;
      loaded.muted = true;
      loaded.playsInline = true;
      if (decode && loaded.readyState >= 2) {
        cell.dataset.frameReady = 'true';
        loaded.classList.remove('is-awaiting-frame');
        const poster = ensurePoster(cell);
        poster.hidden = true;
        poster.classList.remove('is-startup-placeholder');
      }
    }

    function warmAround(index) {
      const generation = ++warmGeneration;
      if (document.body.dataset.page !== 'reels') return;
      /* Keep only three decoder-ready surfaces: current + immediate next/prev.
         +/-2 gets metadata/poster only. This mirrors proximity-ranked preload
         without decoding the entire feed or wasting mobile bandwidth. */
      const next = cells[index + 1], prev = cells[index - 1];
      if (next) preloadNeighbor(next, 'decode');
      if (prev) preloadNeighbor(prev, 'decode');
      const work = function () {
        if (generation !== warmGeneration || document.body.dataset.page !== 'reels') return;
        [cells[index + 2], cells[index - 2]].forEach(function (cell) { if (cell) preloadNeighbor(cell, 'metadata'); });
      };
      if (typeof requestIdleCallback === 'function') requestIdleCallback(work, { timeout: 180 });
      else setTimeout(work, 0);
    }

    function releaseFarVideos(index) {
      cells.forEach(function (cell, cellIndex) {
        if (Math.abs(cellIndex - index) <= 2) return;
        clearVideoSource(cell);
      });
    }

    function updateGlobalReelUI(reel) {
      if (!reel) return;
      const comments = Array.isArray(reel.comments) ? reel.comments : [];
      try { renderReelComments(root, comments); } catch (_error) {}
      const owner = Boolean(profile && String(reel.userId) === String(profile.id));
      root.querySelectorAll('[data-reel-option="edit"],[data-reel-option="delete"]').forEach(function (option) { option.hidden = !owner; });
      if (!owner) root.querySelector('.reels-options')?.classList.remove('open');
      const form = root.querySelector('.reels-comment-form');
      if (form) form.querySelectorAll('input,button').forEach(function (control) { control.disabled = reel.allowComments === false; });
    }

    function playActiveVideo(cell, reset) {
      const video = loadVideoSource(cell, 'auto');
      if (!video) return;
      video.dataset.neighborWarm = '';
      video.playbackRate = 1;
      video.defaultPlaybackRate = 1;
      const wantedMuted = Boolean(nativeMuted);
      video.muted = wantedMuted;
      syncMuteButton(cell);
      if (reset && Number(video.currentTime) > 0.25) {
        try { video.currentTime = 0; } catch (_error) {}
      }
      /* loadLatestReel() also runs during app startup so cached Reels are ready.
       * Preparing the feed must never start playback while Home/Profile/etc. is
       * the visible page. */
      if (document.body.dataset.page !== 'reels' || document.body.classList.contains('reel-create-open') ||
          (pendingReelOpen && !reelMatchesPendingOpen(currentReel))) {
        try { video.pause(); } catch (_error) {}
        return;
      }

      /* A freshly uploaded source can be attached only milliseconds before this
         call. Also, after a long upload the original user activation may have
         expired, so an unmuted autoplay can be rejected by Chromium. Production
         feeds retry when the first frame becomes playable and fall back to muted
         autoplay rather than leaving a dead "tap play again" surface. */
      const token = String((Number(video.dataset.autoplayToken) || 0) + 1);
      video.dataset.autoplayToken = token;
      const reelId = String(cell && cell.dataset.reelId || '');
      const stillCurrent = function () {
        return String(video.dataset.autoplayToken || '') === token &&
          cell.classList.contains('is-active') &&
          String(cell.dataset.reelId || '') === reelId &&
          document.body.dataset.page === 'reels' &&
          !document.body.classList.contains('reel-create-open');
      };
      const waitUntilPlayable = function (timeout) {
        if (video.readyState >= 2) return Promise.resolve();
        return new Promise(function (resolve) {
          let done = false;
          const finish = function () {
            if (done) return;
            done = true;
            video.removeEventListener('loadeddata', finish);
            video.removeEventListener('canplay', finish);
            clearTimeout(timer);
            resolve();
          };
          video.addEventListener('loadeddata', finish, { once:true });
          video.addEventListener('canplay', finish, { once:true });
          const timer = setTimeout(finish, timeout);
        });
      };
      (async function startActiveReel() {
        let forcedMuted = false;
        const waits = [0, 70, 160, 350, 700];
        for (let attemptIndex = 0; attemptIndex < waits.length; attemptIndex += 1) {
          if (!stillCurrent()) return;
          if (waits[attemptIndex]) await new Promise(function(resolve){ setTimeout(resolve, waits[attemptIndex]); });
          if (!stillCurrent()) return;
          await waitUntilPlayable(attemptIndex < 2 ? 350 : 650);
          if (!stillCurrent()) return;
          try {
            video.muted = forcedMuted ? true : wantedMuted;
            await video.play();
            if (!stillCurrent()) return;
            video.dataset.autoplayPending = '';
            /* If autoplay only succeeded because we muted it, try to restore the
               user's previous sound state. If Chromium pauses it when unmuted,
               fall back to muted playback and make the global state truthful. */
            if (forcedMuted && !wantedMuted) {
              video.muted = false;
              await new Promise(function(resolve){ setTimeout(resolve, 90); });
              if (!stillCurrent()) return;
              if (video.paused) {
                video.muted = true;
                nativeMuted = true;
                root.__reelsMuted = true;
                syncMuteButton(cell);
                try { await video.play(); } catch (_error) {}
              }
            }
            return;
          } catch (error) {
            if (!stillCurrent()) return;
            const name = String(error && error.name || '');
            if (!wantedMuted && !forcedMuted && (name === 'NotAllowedError' || attemptIndex >= 1)) {
              forcedMuted = true;
              continue;
            }
            video.dataset.autoplayPending = '1';
          }
        }
      })();
    }

    function activateIndex(index, options) {
      if (!reelFeed.length || !cells.length) return;
      index = Math.max(0, Math.min(reelFeed.length - 1, Number(index) || 0));
      const changed = index !== activeIndex;
      const previous = activeIndex;
      activeIndex = index;
      currentReelIndex = index;
      currentReel = reelFeed[index] || null;
      cells.forEach(function (cell, cellIndex) {
        const isActive = cellIndex === index;
        cell.classList.toggle('is-active', isActive);
        cell.setAttribute('aria-hidden', isActive ? 'false' : 'true');
        const video = cellVideo(cell);
        if (!isActive && video) {
          try { video.pause(); } catch (_error) {}
          video.muted = true;
          /* The just-posted Reel used the local Blob only for its first view.
             Once it leaves the viewport, release that memory. Returning to the
             Reel will use the normal HTTP-range server URL like every other Reel. */
          if (cellIndex === previous && cell.__reel) {
            const immediate = reelImmediatePlaybackUrls.get(String(cell.__reel.id || ''));
            if (immediate) {
              if (String(video.dataset.loadedSource || '') === immediate) clearVideoSource(cell);
              releaseImmediateReelPlaybackUrl(cell.__reel.id);
              if (String(cell.__reel.cachedVideo || '') === immediate) delete cell.__reel.cachedVideo;
            }
          }
        }
      });
      const activeCell = cells[index];
      syncActiveIds(activeCell);
      updateGlobalReelUI(currentReel);
      playActiveVideo(activeCell, changed && previous >= 0);
      if (currentReel && document.body.dataset.page === 'reels') scheduleReelView(currentReel, cellVideo(activeCell));
      if (!(options && options.deferWarm)) warmAround(index);
      if (!(options && options.deferRelease)) releaseFarVideos(index);
      loadMoreReelsIfNeeded(index);
    }

    function buildSignature() {
      return reelFeed.map(function (reel) { return String(reel && reel.id || '') + '@' + String(reel && reel.video || ''); }).join('|');
    }

    function rebuildNativeFeed() {
      if (!root.__nativeReelsActive) return;
      const signature = buildSignature();
      if (!reelFeed.length) {
        cells.forEach(function (cell) { clearVideoSource(cell); });
        feedSignature = '';
        activeIndex = -1;
        return;
      }
      root.__nativeReelsPainting = true;
      try {
        /* Pagination must never tear down the currently playing cell. If the
           new feed is simply the old prefix plus more Reels, append only the
           new native cells and keep every prepared/current video element alive. */
        const appendOnly = cells.length > 0 && cells.length < reelFeed.length && cells.every(function (cell, index) {
          return cell && cell.__reel && reelFeed[index] && String(cell.__reel.id || '') === String(reelFeed[index].id || '');
        });
        if (appendOnly) {
          const start = cells.length;
          while (cells.length < reelFeed.length) {
            const clone = prototype.cloneNode(true);
            clone.dataset.nativeOriginal = 'false';
            clone.querySelectorAll('[id]').forEach(function (node) { node.removeAttribute('id'); });
            clone.querySelectorAll('video').forEach(function (video) {
              video.removeAttribute('src');
              video.removeAttribute('poster');
              video.dataset.loadedSource = '';
              video.dataset.neighborWarm = '';
            });
            markStableRoles(clone);
            wireCell(clone);
            scroller.appendChild(clone);
            cells.push(clone);
          }
          for (let index = start; index < cells.length; index += 1) paintCell(cells[index], reelFeed[index], index);
          feedSignature = signature;
        } else if (signature !== feedSignature || cells.length !== reelFeed.length) {
          const firstVideo = cellVideo(firstCell);
          const preserveFirst = activeIndex === 0 && firstCell && firstCell.__reel && reelFeed[0] &&
            String(firstCell.__reel.id || '') === String(reelFeed[0].id || '');
          const preservedTime = preserveFirst && firstVideo ? Number(firstVideo.currentTime || 0) : 0;
          const preservedPaused = preserveFirst && firstVideo ? Boolean(firstVideo.paused) : true;
          cells.forEach(function (cell, cellIndex) {
            if (preserveFirst && cellIndex === 0) return;
            const video = cellVideo(cell);
            if (video) { try { video.pause(); } catch (_error) {} }
          });
          /* Never detach the active first video when only the surrounding feed
           * changed. Android Chromium can destroy/recreate the video compositor
           * surface on detach, producing the visible shrink/pause/restart. */
          if (preserveFirst && firstCell.parentNode === scroller) {
            Array.from(scroller.children).forEach(function (child) {
              if (child !== firstCell) child.remove();
            });
          } else {
            scroller.replaceChildren(firstCell);
          }
          cells = [firstCell];
          while (cells.length < reelFeed.length) {
            const clone = prototype.cloneNode(true);
            clone.dataset.nativeOriginal = 'false';
            clone.querySelectorAll('[id]').forEach(function (node) { node.removeAttribute('id'); });
            clone.querySelectorAll('video').forEach(function (video) {
              video.removeAttribute('src');
              video.removeAttribute('poster');
              video.dataset.loadedSource = '';
              video.dataset.neighborWarm = '';
            });
            markStableRoles(clone);
            wireCell(clone);
            scroller.appendChild(clone);
            cells.push(clone);
          }
          feedSignature = signature;
          if (!preserveFirst) activeIndex = -1;
          else {
            activeIndex = 0;
            currentReelIndex = 0;
            currentReel = reelFeed[0] || currentReel;
            if (firstVideo) {
              if (preservedTime > 0 && Math.abs(Number(firstVideo.currentTime || 0) - preservedTime) > .15) {
                try { firstVideo.currentTime = preservedTime; } catch (_error) {}
              }
              if (!preservedPaused && document.body.dataset.page === 'reels') {
                const resumed = firstVideo.play();
                if (resumed && typeof resumed.catch === 'function') resumed.catch(function () {});
              }
            }
          }
        }
        if (!appendOnly) cells.forEach(function (cell, index) { paintCell(cell, reelFeed[index], index); });
      } finally {
        root.__nativeReelsPainting = false;
      }
      const wanted = Math.max(0, Math.min(reelFeed.length - 1, currentReelIndex || 0));
      activateIndex(wanted, { deferRelease: true });
      requestAnimationFrame(function () {
        const cell = cells[wanted];
        if (cell) scroller.scrollTop = cell.offsetTop;
        releaseFarVideos(wanted);
      });
    }
